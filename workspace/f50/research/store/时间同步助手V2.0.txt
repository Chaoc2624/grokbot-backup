  function smartMapUrl(nameOrUrl) {
    var v = String(nameOrUrl || "").trim();
    if (!v) return "http://www.baidu.com";

    // 用户已填完整 URL，尊重用户
    if (v.indexOf("http://") === 0 || v.indexOf("https://") === 0) return v;

    var low = v.toLowerCase();

    // ✅ NTP 域名不是网页服务器：映射到可返回 Date 的网站
    // 优先 http 作为第一跳（避免时间错导致 TLS 各种拦截）
    if (low === "ntp.aliyun.com" || low.indexOf("aliyun") >= 0) return "http://www.aliyun.com";
    if (low === "ntp.tencent.com" || low.indexOf("tencent") >= 0 || low.indexOf("qq.com") >= 0) return "http://www.qq.com";
    if (low === "ntp.ntsc.ac.cn" || low.indexOf("ntsc") >= 0) return "http://www.ntsc.ac.cn";

    // 其它域名：默认 https（有 -k + retry 兜底）
    return "https://" + v;
  }

  function swapProtocol(url) {
    if (url.indexOf("https://") === 0) return "http://" + url.slice(8);
    if (url.indexOf("http://") === 0) return "https://" + url.slice(7);
    return url;
  }

  function uniq(arr) {
    var m = {};
    var out = [];
    for (var i = 0; i < arr.length; i++) {
      var x = String(arr[i] || "").trim();
      if (!x) continue;
      if (m[x]) continue;
      m[x] = 1;
      out.push(x);
    }
    return out;
  }

  }

  /* ================== 同步逻辑（手动/自动共用） ================== */
  async function syncOnce(mode) {
    var modeCn = (mode === "boot") ? "自动" : "手动";

    if (!(await lockAcquire())) {
      toast("正在同步中，请稍后再试…", "red");
      logResult(modeCn, "-", "失败", "同步被拒绝：已有同步任务在运行");
      return false;
    }

    try {
      logResult(modeCn, "-", "开始", "开始同步（顺序：1→2→3）");
      logProc(modeCn + "同步开始（顺序：1→2→3）");

      var curl = "curl";
      var ck = await runRoot("command -v curl || [ -x /data/data/com.minikano.f50_sms/files/curl ] && echo /data/data/com.minikano.f50_sms/files/curl");
      if (ck.content) curl = ck.content.trim();

      for (var i = 0; i < SERVERS.length; i++) {
        var idx = i + 1;
        var s = SERVERS[i];

        var urls = uniq([s.url, swapProtocol(s.url)]);

        for (var u = 0; u < urls.length; u++) {
          var url = urls[u];

          logProc("尝试第" + idx + "号服务器：" + s.name + "（" + url + "）");

          var r = await runRoot(curl + " " + CURL_COMMON + " '" + url + "' | grep -i '^Date:' | head -n1");

          var code = await runRoot(curl + " -m 10 -s -k " + CURL_RETRY + " -o /dev/null -w '%{http_code}' '" + url + "' 2>/dev/null || echo 000");
          var httpCode = (code.content || "").trim() || "000";

          if (!r.content) {
            logResult(modeCn, s.name, "失败", "未获取到时间（http_code=" + httpCode + "；网络未通/握手失败/无 Date 头）");
            continue;
          }

          var dateStr = "";
          var p = String(r.content).indexOf(":");
          if (p >= 0) dateStr = String(r.content).slice(p + 1).trim();

          var d = Date.parse(dateStr);
          if (!d) {
            logResult(modeCn, s.name, "失败", "时间解析失败（Date 头异常；http_code=" + httpCode + "）");
            continue;
          }

          var t = new Date(d);
          var utc =
            t.getUTCFullYear() + "-" + pad(t.getUTCMonth() + 1) + "-" + pad(t.getUTCDate()) + " " +
            pad(t.getUTCHours()) + ":" + pad(t.getUTCMinutes()) + ":" + pad(t.getUTCSeconds());

          var ok = await runRoot("date -u -s '" + utc + "' >/dev/null 2>&1; echo $?");
          if ((ok.content || "").trim() === "0") {
            var detail = "与第" + idx + "号服务器同步成功";
            logResult(modeCn, s.name, "成功", detail + "（http_code=" + httpCode + "）");
            logProc("同步成功！" + detail + "。同步地址：" + s.name);

            saveLastSync(fmtLocal(new Date()), modeCn);
            updateLastSyncUI();

            if (mode !== "boot") toast("同步成功！同步地址：" + s.name, "green");
            return true;
          } else {
            logResult(modeCn, s.name, "失败", "设置系统时间失败（权限/系统限制；http_code=" + httpCode + "）");
          }
        }
      }

      logResult(modeCn, "-", "失败", "所有服务器均失败");
      logProc("同步失败：1/2/3号服务器均失败（检查网络/curl/权限/地址）");
      if (mode !== "boot") toast("同步失败，请查看日志", "red");
      return false;

    } finally {
      await lockRelease();
    }
  }

  /* ================== 页面加载时用日志回填“自动同步”状态 ================== */
  function parseLatestAutoSuccessFromRaw(raw) {
    if (!raw) return null;
    var lines = raw.split("\n");
    for (var i = lines.length - 1; i >= 0; i--) {
      var line = String(lines[i] || "").trim();
      if (line.indexOf("[TSRES] ") !== 0) continue;
      if (line.indexOf(" mode=自动 ") < 0) continue;
      if (line.indexOf(" result=成功 ") < 0) continue;
      var ts = line.substring(8).trim().substring(0, 19);
      if (ts && ts.length === 19) return { ts: ts, mode: "自动" };
    }
    return null;
  }

  async function reconcileAutoFromLog() {
    var raw = await readLogRaw();
    var rec = parseLatestAutoSuccessFromRaw(raw);
    if (!rec) return;
    if (!lastSync.ts || rec.ts > lastSync.ts) {
      saveLastSync(rec.ts, "自动");
      updateLastSyncUI();
    }
  }

  /* ================== 日志弹窗（固定高度可滚动） ================== */
  function parseLog(raw) {
    var rowsRes = [];
    var rowsProc = [];
    if (!raw) return { res: rowsRes, proc: rowsProc };

    var lines = raw.split("\n");
    for (var i = 0; i < lines.length; i++) {
      var line = String(lines[i] || "").trim();

      if (line.indexOf("[TSRES] ") === 0) {
        var rest = line.substring(8).trim();
        var ts = rest.length >= 19 ? rest.substring(0, 19) : "";
        var tail = rest.length > 19 ? rest.substring(19).trim() : "";

        function pick(key) {
          var p = tail.indexOf(key + "=");
          if (p < 0) return "";
          var sub = tail.substring(p + key.length + 1);
          var cut = sub.length;
          var keys = ["mode=", "server=", "result=", "detail="];
          for (var k = 0; k < keys.length; k++) {
            var kk = keys[k];
            if (kk === key + "=") continue;
            var j = sub.indexOf(" " + kk);
            if (j >= 0 && j < cut) cut = j;
          }
          return sub.substring(0, cut).trim();
        }

        rowsRes.push({
          ts: ts,
          mode: pick("mode"),
          server: pick("server"),
          result: pick("result"),
          detail: pick("detail")
        });

      } else if (line.indexOf("[TSPROC] ") === 0) {
        var r2 = line.substring(9).trim();
        var ts2 = r2.length >= 19 ? r2.substring(0, 19) : "";
        var tail2 = r2.length > 19 ? r2.substring(19).trim() : "";
        var text = "";
        var p2 = tail2.indexOf("text=");
        if (p2 >= 0) text = tail2.substring(p2 + 5);
        rowsProc.push({ ts: ts2, text: text });
      }
    }

    return { res: rowsRes, proc: rowsProc };
  }

  /* ================== boot.sh（保持块替换 + 同样的智能映射） ================== */
  function bootBlockText() {
    function mapToWeb(v) {
      v = String(v || "").trim();
      if (!v) return "http://www.baidu.com";
      if (v.indexOf("http://") === 0 || v.indexOf("https://") === 0) return v;
      var low = v.toLowerCase();
      if (low === "ntp.aliyun.com" || low.indexOf("aliyun") >= 0) return "http://www.aliyun.com";
      if (low === "ntp.tencent.com" || low.indexOf("tencent") >= 0 || low.indexOf("qq.com") >= 0) return "http://www.qq.com";
      if (low === "ntp.ntsc.ac.cn" || low.indexOf("ntsc") >= 0) return "http://www.ntsc.ac.cn";
      return "https://" + v;
    }

    var s1 = String(serverNames[0] || "").trim();
    var s2 = String(serverNames[1] || "").trim();
    var s3 = String(serverNames[2] || "").trim();
    var u1 = mapToWeb(s1);
    var u2 = mapToWeb(s2);
    var u3 = mapToWeb(s3);

    var CURL_PATH = "/data/data/com.minikano.f50_sms/files/curl";

    var blk =
      BOOT_BEGIN + "\n" +
      "(\n" +
      "  LOG_FILE='" + LOG_FILE + "'\n" +
      "  LOCK_DIR='" + LOCK_DIR + "'\n" +
      "  mkdir -p /data/local/tmp >/dev/null 2>&1 || true\n" +
      "  rm -f \"$LOG_FILE\" 2>/dev/null || true\n" +
      "  rmdir \"$LOCK_DIR\" 2>/dev/null || true\n" +
      "  sleep " + BOOT_INITIAL_DELAY_SEC + "\n" +
      "  now(){ date '+%Y-%m-%d %H:%M:%S'; }\n" +
      "  wres(){ echo \"[TSRES] $(now) mode=自动 server=$1 result=$2 detail=$3\" >> \"$LOG_FILE\" 2>/dev/null || true; }\n" +
      "  wproc(){ echo \"[TSPROC] $(now) text=$1\" >> \"$LOG_FILE\" 2>/dev/null || true; }\n" +
      "  mkdir \"$LOCK_DIR\" 2>/dev/null || { wres '-' '失败' '同步被拒绝：已有同步任务在运行'; exit 0; }\n" +
      "  trap 'rmdir \"$LOCK_DIR\" 2>/dev/null || true' EXIT\n" +
      "  CURL=curl\n" +
      "  [ -x \"" + CURL_PATH + "\" ] && CURL=\"" + CURL_PATH + "\"\n" +
      "  wres '-' '开始' '开始开机自动同步（带重试）'\n" +
      "  wproc '开机自动同步开始（带重试）'\n" +
      "  mon2num(){\n" +
      "    case \"$1\" in\n" +
      "      Jan) echo 01;; Feb) echo 02;; Mar) echo 03;; Apr) echo 04;; May) echo 05;; Jun) echo 06;;\n" +
      "      Jul) echo 07;; Aug) echo 08;; Sep) echo 09;; Oct) echo 10;; Nov) echo 11;; Dec) echo 12;;\n" +
      "      *) echo 00;;\n" +
      "    esac\n" +
      "  }\n" +
      "  rfc1123_to_utc(){\n" +
      "    s=\"$1\"\n" +
      "    s=$(echo \"$s\" | sed 's/^[Dd]ate:[ ]*//')\n" +
      "    s=$(echo \"$s\" | sed 's/,//')\n" +
      "    set -- $s\n" +
      "    d=\"$2\"; m=\"$3\"; y=\"$4\"; t=\"$5\"\n" +
      "    mm=$(mon2num \"$m\")\n" +
      "    [ \"$mm\" = \"00\" ] && echo '' && return\n" +
      "    [ ${#d} -eq 1 ] && d=\"0$d\"\n" +
      "    echo \"$y-$mm-$d $t\"\n" +
      "  }\n" +
      "  swap_proto(){\n" +
      "    x=\"$1\"\n" +
      "    echo \"$x\" | grep -q '^https://' && echo \"http://$(echo $x | cut -c9-)\" && return\n" +
      "    echo \"$x\" | grep -q '^http://' && echo \"https://$(echo $x | cut -c8-)\" && return\n" +
      "    echo \"$x\"\n" +
      "  }\n" +
      "  try_url(){\n" +
      "    idx=\"$1\"; name=\"$2\"; url=\"$3\"\n" +
      "    wproc \"尝试第${idx}号服务器：${name}（${url}）\"\n" +
      "    DATE_LINE=$($CURL " + CURL_COMMON + " \"$url\" | grep -i '^Date:' | head -n1)\n" +
      "    code=$($CURL -m 10 -s -k " + CURL_RETRY + " -o /dev/null -w '%{http_code}' \"$url\" 2>/dev/null || echo 000)\n" +
      "    [ -z \"$DATE_LINE\" ] && wres \"$name\" '失败' \"未获取到时间（http_code=${code}；网络未通/握手失败/无 Date 头）\" && return 1\n" +
      "    UTC=$(rfc1123_to_utc \"$DATE_LINE\")\n" +
      "    [ -z \"$UTC\" ] && wres \"$name\" '失败' \"时间解析失败（Date 头异常；http_code=${code}）\" && return 1\n" +
      "    date -u -s \"$UTC\" >/dev/null 2>&1\n" +
      "    if [ \"$?\" = \"0\" ]; then\n" +

      "    fi\n" +
      "  }\n" +
      "  try_one(){\n" +
      "    idx=\"$1\"; name=\"$2\"; url=\"$3\"\n" +
      "    try_url \"$idx\" \"$name\" \"$url\" && return 0\n" +
      "    try_url \"$idx\" \"$name\" \"$(swap_proto \"$url\")\" && return 0\n" +
      "    return 1\n" +

      "  MAX=" + BOOT_RETRY_MAX + "\n" +
      "  BASE=" + BOOT_RETRY_BASE_INTERVAL_SEC + "\n" +

      "    wproc \"第${i}次自动同步：开始（顺序 1→2→3）\"\n" +
      "    try_one 1 '" + s1 + "' '" + u1 + "' && exit 0\n" +
      "    try_one 2 '" + s2 + "' '" + u2 + "' && exit 0\n" +
      "    try_one 3 '" + s3 + "' '" + u3 + "' && exit 0\n" +
      "    wproc \"第${i}次自动同步：失败\"\n" +
      "    if [ \"$i\" -lt \"$MAX\" ]; then\n" +
      "      wait=$((BASE + (i-1)*10))\n" +
      "      wproc \"等待 ${wait} 秒后重试\"\n" +
      "      sleep \"$wait\"\n" +
      "    fi\n" +
      "    i=$((i+1))\n" +
      "  done\n" +
      "  wres '-' '失败' '开机自动同步：多次重试仍失败（检查网络/DNS/权限）'\n" +
      "  wproc '开机自动同步：已达到最大重试次数'\n" +
      ") >/dev/null 2>&1 &\n" +
      BOOT_END + "\n";

    return blk;
  }

  async function bootEnabled() {
    var r = await runRoot("grep -F '" + BOOT_BEGIN + "' '" + BOOT_SH + "' >/dev/null 2>&1; echo $?");
    return (r.content || "").trim() === "0";
  }

  }

  async function setBootEnabled(enable) {
    await runRoot("mkdir -p /sdcard >/dev/null 2>&1 || true; [ -f '" + BOOT_SH + "' ] || touch '" + BOOT_SH + "'");

    await runRoot("sed -i '/" + BOOT_BEGIN.replace(/#/g, "\\#") + "/,/" + BOOT_END.replace(/#/g, "\\#") + "/d' '" + BOOT_SH + "' 2>/dev/null || true");

    if (enable) {
      var blk = bootBlockText();
      await runRoot("printf '%s' '" + esc(blk) + "' >> '" + BOOT_SH + "' 2>/dev/null || true");
      logResult("自动", "-", "配置", "已开启开机自同步");
      logProc("已开启开机自同步（写入 boot.sh 块）");
      toast("已开启开机自同步", "green");
    } else {
      logResult("自动", "-", "配置", "已关闭开机自同步");
      logProc("已关闭开机自同步（移除 boot.sh 块）");
      toast("已关闭开机自同步", "green");
    }

    await refreshBootUI();
  }
